# SpectreTTS — requirements.txt
# Active backend: pocket-tts (Kyutai) — replaces kokoro-onnx.
#
# Install inside the project venv (Python 3.14+):
#   source .venv/bin/activate
#   pip install -r requirements.txt
#
# System packages (NOT pip-installable, install via apt first):
#   libportaudio2 xclip xdotool python3-gi python3-gi-cairo
#   gir1.2-gtk-3.0 libayatana-appindicator3-dev gir1.2-ayatanaappindicator3-0.1
#   libgirepository-2.0-dev pkg-config python3-dev gcc

# ── TTS engine ──────────────────────────────────────────────────────────────
pocket-tts>=3.3.0            # Kyutai pocket-sized TTS — streaming, CPU-capable, auto-downloads models
piper-tts>=1.2.0             # Piper TTS — lightweight, ONNX-based, no PyTorch required
huggingface-hub>=0.20.0      # Required by Piper backend to auto-download voice models

# ── Config ──────────────────────────────────────────────────────────────────
python-dotenv>=1.0.0         # Load .env file for configuration

# ── Audio ───────────────────────────────────────────────────────────────────
soundfile>=0.12.1
sounddevice>=0.4.7

# ── Systray + GTK UI ────────────────────────────────────────────────────────
PyGObject>=3.48.0            # GTK3 + AyatanaAppIndicator3 bindings

# ── Notes ───────────────────────────────────────────────────────────────────
# - PyGObject requires system dev headers (libgirepository-2.0-dev, libcairo2-dev,
#   pkg-config, python3-dev) present BEFORE `pip install` — see README.md.
# - pocket-tts pulls in PyTorch (CPU or CUDA depending on platform). Model weights
#   are auto-downloaded from HuggingFace on first use (~225MB english model).
# - Set SPECTRETTS_POCKET_LANGUAGE env var to change language (default: english).
#   Available: english, french, german, portuguese, italian, spanish.
